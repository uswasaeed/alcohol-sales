Convert data types

Fixing the data types.

bev_clean <- bev_col |>
  mutate(
    date = mdy(obligation_end_date),
    year = year(date),
    location_zip = as.character(location_zip),
    beer_receipts = as.numeric(gsub("[,$]", "", beer_receipts)),
    wine_receipts = as.numeric(gsub("[,$]", "", wine_receipts)),
    liquor_receipts = as.numeric(gsub("[,$]", "", liquor_receipts)),
    total_receipts = as.numeric(gsub("[,$]", "", total_receipts))
  )
